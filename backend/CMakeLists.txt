cmake_minimum_required(VERSION 3.20)
project(VehicleParkingSystem LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Drogon CONFIG REQUIRED)
add_executable(parking_api src/main.cpp src/availability.cpp)
target_link_libraries(parking_api PRIVATE Drogon::Drogon)
if(MSVC)
  target_compile_options(parking_api PRIVATE /W4)
else()
  target_compile_options(parking_api PRIVATE -Wall -Wextra -Wpedantic)
endif()
